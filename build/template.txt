function frFields(off){

  const { P, n, nx, ny, W } = frw, cells = nx*ny;
  const code = new Array(n);
  for (let p=0;p<n;p++) code[p] = (P[p][2] <= off && off <= P[p][3]) ? P[p].map.get(off) : undefined;
  const tv = new Float64Array(n);
  for (let p=0;p<n;p++){ const k = off - P[p][2]; tv[p] = (code[p] && P[p].temp && k >= 0 && k < P[p].temp.length) ? P[p].temp[k] - 60 : NaN; }
  const lo = new Float64Array(n), hi = new Float64Array(n), rc = new Float64Array(n), rh = new Float64Array(n);
  for (let p=0;p<n;p++){ lo[p] = P[p].lo.get(off) ?? 5; hi[p] = P[p].hi.get(off) ?? 95; rc[p] = (P[p].rc.get(off) ?? 0)/20; rh[p] = (P[p].rh.get(off) ?? 0)/20; }
  const tier = new Float64Array(cells), mild = new Float64Array(cells), temp = new Float64Array(cells);
  const arctic = new Float64Array(cells), heat = new Float64Array(cells), rareC = new Float64Array(cells), rareH = new Float64Array(cells);
  for (let q=0;q<cells;q++){
    let den=0, t=0, sp=0, au=0, tw=0, ts=0, sl=0, sh=0, sc=0, su=0; const row=q*n;
    for (let p=0;p<n;p++){
      const c = code[p]; if (!c) continue;
      const w = W[row+p]; den += w; t += w*TIER[c];
      if (c==='s') sp += w; else if (c==='a') au += w;
      sl += w*lo[p]; sh += w*hi[p]; sc += w*rc[p]; su += w*rh[p];
      if (tv[p] === tv[p]){ tw += w; ts += w*tv[p]; }
    }
    tier[q] = den ? t/den : 1; mild[q] = den ? (sp-au)/den : 0; temp[q] = tw ? ts/tw : 55;
    arctic[q] = den ? -sl/den : -5; heat[q] = den ? sh/den : 95;
    rareC[q] = den ? sc/den : 0; rareH[q] = den ? su/den : 0;

const MIN_AREA = 900;            // px²: patches and holes smaller than this are dropped so regions stay contiguous
const ringArea = r => { let a = 0; for (let i=0, j=r.length-1; i<r.length; j=i++) a += (r[j][0]+r[i][0])*(r[j][1]-r[i][1]); return Math.abs(a/2); };
function ringPaths(mp){
  const k = frw.SK, rings = [];
  for (const poly of mp.coordinates){
    const outer = poly[0].map(p=>[p[0]*k, p[1]*k]);
    const minA = [MIN_AREA, 160, 30][sample(frw.cellReg, outer[0][0], outer[0][1])];   // Alaska and Hawaii are drawn small
    if (ringArea(outer) < minA) continue;
    rings.push(outer);
    for (const hole of poly.slice(1)){ const h = hole.map(p=>[p[0]*k, p[1]*k]); if (ringArea(h) >= minA) rings.push(h); }
  }
  return rings;
}
const toD = rings => rings.map(r=>'M'+r.map(p=>p[0].toFixed(1)+','+p[1].toFixed(1)).join('L')+'Z').join('');
const lineD = pts => 'M'+pts.map(p=>p[0].toFixed(1)+','+p[1].toFixed(1)).join('L');
function sample(arr, x, y){
  const i = Math.max(0, Math.min(frw.nx-1, Math.floor(x/frw.SK))), j = Math.max(0, Math.min(frw.ny-1, Math.floor(y/frw.SK)));
  return arr[j*frw.nx+i];
}
// Cut each front ring into ~34px pieces, decide for each whether the cold or warm side is advancing, smooth those
// decisions so fronts come in long steady stretches, then draw one line per stretch with its symbols.
// Start every ring at the same place (its westernmost point), so symbols spaced along it stay put from frame to
// frame instead of hopping whenever the contour happens to begin somewhere else.
function anchorRing(r){
  const closed = r.length > 1 && r[0][0] === r[r.length-1][0] && r[0][1] === r[r.length-1][1];
  const pts = closed ? r.slice(0, -1) : r;
  let k = 0; for (let i=1;i<pts.length;i++) if (pts[i][0] < pts[k][0] - 1e-6 || (Math.abs(pts[i][0]-pts[k][0]) < 1e-6 && pts[i][1] < pts[k][1])) k = i;
  const out = pts.slice(k).concat(pts.slice(0, k));
  out.push(out[0]);
  return out;
}
// Walk a ring by distance: evenly spaced pieces whose ends and middles sit at exact distances along the line
function arcPieces(r, STEP){
  const cum = [0]; for (let i=1;i<r.length;i++) cum.push(cum[i-1] + Math.hypot(r[i][0]-r[i-1][0], r[i][1]-r[i-1][1]));
  const L = cum[cum.length-1]; if (!(L > 0)) return [];
  let seg = 0;
  const at = s => { s = Math.max(0, Math.min(L, s)); while (seg > 0 && cum[seg] > s) seg--; while (seg < r.length-2 && cum[seg+1] < s) seg++;
    const a = r[seg], b = r[seg+1], len = cum[seg+1]-cum[seg] || 1, u = (s - cum[seg]) / len; return [a[0] + (b[0]-a[0])*u, a[1] + (b[1]-a[1])*u]; };
  // fixed spacing from the anchor; the leftover bit at the end is one shorter piece, so a growing front only
  // moves its last symbol instead of re-spacing all of them
  const cuts = []; for (let x=0; x < L - 1e-6; x += STEP) cuts.push(x); cuts.push(L);
  if (cuts.length > 2 && cuts[cuts.length-1] - cuts[cuts.length-2] < STEP*0.35) cuts.splice(cuts.length-2, 1);
  const pieces = [];
  let vi = 1;
  for (let j=0;j<cuts.length-1;j++){
    const a = cuts[j], b = cuts[j+1], mid = a + Math.min(STEP, b - a)/2;
    const pts = [at(a)];
    while (vi < r.length-1 && cum[vi] <= a) vi++;
    for (let k=vi; k<r.length-1 && cum[k] < b; k++) pts.push(r[k]);
    pts.push(at(b));
    const m = at(mid), p0 = at(mid - 6), p1 = at(mid + 6);
    let tx = p1[0]-p0[0], ty = p1[1]-p0[1]; const tl = Math.hypot(tx, ty) || 1;
    pieces.push(Object.assign(pts, { m, tx: tx/tl, ty: ty/tl }));
  }
  return pieces;
}
function frontPieces(rings, tier, dT, out){
  for (let r of rings){
    if (r.length < 3) continue;
    r = anchorRing(r);
    const K = REGION_K[sample(frw.cellReg, r[0][0], r[0][1])];
    const pieces = arcPieces(r, 46*K);
    if (!pieces.length) continue;
    for (const pc of pieces){
      const { m, tx, ty } = pc;
      let wx = -ty, wy = tx;
      if (sample(tier, m[0]+wx*8, m[1]+wy*8) < sample(tier, m[0]-wx*8, m[1]-wy*8)){ wx = -wx; wy = -wy; }   // (wx,wy) -> warm side
      Object.assign(pc, { wx, wy, motion: sample(dT, m[0], m[1]) });
    }
    // smooth motion along the ring (5-piece window), then classify with a dead band
    const n = pieces.length;
    const sm = pieces.map((_, i)=>{ let s = 0, c = 0; for (let d=-2; d<=2; d++){ const q = pieces[(i+d+n)%n]; s += q.motion; c++; } return s/c; });
    let types = sm.map(v => v < -0.012 ? 'cold' : v > 0.012 ? 'warm' : 'still');
    // absorb runs shorter than 3 pieces into their neighbours
    for (let pass=0; pass<2; pass++){
      for (let i=0;i<n;i++){
        let j = i; while (j+1 < n && types[j+1] === types[i]) j++;
        if (j-i+1 < 3 && n > 3){ const nb = types[(i-1+n)%n] !== types[i] ? types[(i-1+n)%n] : types[(j+1)%n]; for (let k=i;k<=j;k++) types[k] = nb; }
        i = j;
      }
    }
    // a lone stalled piece between moving stretches takes its neighbours' type
    for (let i=0;i<n;i++){
      if (types[i] !== 'still') continue;
      const pa = types[(i-1+n)%n], pb = types[(i+1)%n];
      if (pa !== 'still' && pb !== 'still') types[i] = pa;
    }
    // join consecutive pieces of one type into a single stretch
    let i = 0;
    while (i < n){
      let j = i; while (j+1 < n && types[j+1] === types[i]) j++;
      const pts = []; for (let k=i;k<=j;k++){ const pc = pieces[k]; for (let q=(k===i?0:1); q<pc.length; q++) pts.push(pc[q]); }
      out.lines.push({ type: types[i], d: lineD(pts), k: K });
      for (let k=i;k<=j;k++){
        const { m, tx, ty, wx, wy } = pieces[k];
        if (types[i] === 'cold'){        // triangle toward the warm side: the way the cold air is moving
          out.syms.push({ type: 'cold', k: K, d: `M${(m[0]-tx*10*K).toFixed(1)},${(m[1]-ty*10*K).toFixed(1)}L${(m[0]+wx*15*K).toFixed(1)},${(m[1]+wy*15*K).toFixed(1)}L${(m[0]+tx*10*K).toFixed(1)},${(m[1]+ty*10*K).toFixed(1)}Z` });
        } else if (types[i] === 'warm'){ // half-circle toward the cold side: the way the warm air is moving
          const ptsC = [];
          for (let s=0;s<=12;s++){ const a = Math.PI*s/12; ptsC.push([m[0] + tx*10*K*Math.cos(a) - wx*10*K*Math.sin(a), m[1] + ty*10*K*Math.cos(a) - wy*10*K*Math.sin(a)]); }
          out.syms.push({ type: 'warm', k: K, d: lineD(ptsC)+'Z' });
        }
      }
      i = j + 1;
    }
  }
}

// Arctic: icicle spikes and snowflakes. Heat: flames and suns. Spikes and flames point the way the edge is moving.
let showArctic = true; const showHeat = false;     // heat fronts are off the map; rare heat alerts stay
function extremeFront(rings, field, dF, kind, out){
  for (let r of rings){
    if (r.length < 3) continue;
    r = anchorRing(r);
    const rg = sample(frw.cellReg, r[0][0], r[0][1]), K = REGION_K[rg], STEP = 46*K;
    const minA = [MIN_AREA, 160, 30][rg], ar = ringArea(r);
    const op = Math.min(1, Math.max(0, (ar - minA) / (5*minA)));     // small areas fade in and out instead of popping
    const s0 = out.syms.length, f0 = out.flakes.length, u0 = out.suns.length;
    let k = 0; const f2 = v=>v.toFixed(1);
    for (const pc of arcPieces(r, STEP)){                        // symbols at exact distances from the anchor
      const { m, tx, ty } = pc;
      let cx = -ty, cy = tx;                                     // (cx,cy) -> toward the core (stronger side)
      if (sample(field, m[0]+cx*8, m[1]+cy*8) < sample(field, m[0]-cx*8, m[1]-cy*8)){ cx = -cx; cy = -cy; }
      const grow = sample(dF, m[0], m[1]) >= -0.3;               // growing or holding: the edge pushes outward
      const dx = grow ? -cx : cx, dy = grow ? -cy : cy;          // the way it is moving
      const P = (u, v)=>[m[0] + (tx*u + dx*v)*K, m[1] + (ty*u + dy*v)*K], pt = q=>f2(q[0])+','+f2(q[1]);
      if (kind === 'arctic'){
        if (k % 2 === 0){                                        // double icicle
          for (const [o, h] of [[-6, 22], [6, 16]]){
            out.syms.push({ type: kind, d: `M${pt(P(o-4.5,0))}L${pt(P(o,h))}L${pt(P(o+4.5,0))}Z` });
          }
        } else {                                                 // snowflake sitting on the line
          let d = '';
          for (let s=0;s<3;s++){ const an = Math.PI*s/3 + 0.26, ux = Math.cos(an)*9*K, uy = Math.sin(an)*9*K;
            d += `M${f2(m[0]-ux)},${f2(m[1]-uy)}L${f2(m[0]+ux)},${f2(m[1]+uy)}`; }
          out.flakes.push({ d, k: K });
        }
      } else {
        if (k % 2 === 0){                                        // flame
          const b1 = P(-9,0), b2 = P(9,0), tip = P(3,24);
          out.syms.push({ type: kind, d: `M${pt(b1)}C${pt(P(-12,9))} ${pt(P(-4,14))} ${pt(P(-2,18))}C${pt(P(-1,13))} ${pt(P(1.5,15))} ${pt(tip)}C${pt(P(8,16))} ${pt(P(13,8))} ${pt(b2)}Z` });
        } else {                                                 // sun
          let rays = '';
          for (let s=0;s<8;s++){ const an = Math.PI*s/4, ux = Math.cos(an), uy = Math.sin(an);
            rays += `M${f2(m[0]+ux*9.5*K)},${f2(m[1]+uy*9.5*K)}L${f2(m[0]+ux*14*K)},${f2(m[1]+uy*14*K)}`; }
          out.suns.push({ c: m, rays, k: K });
        }
      }
      k++;
    }
    for (let q=s0;q<out.syms.length;q++) out.syms[q].op = op;
    for (let q=f0;q<out.flakes.length;q++) out.flakes[q].op = op;
    for (let q=u0;q<out.suns.length;q++) out.suns[q].op = op;
    out.lines.push({ type: kind, d: lineD(r.concat([r[0]])), k: K, op, pts: r.concat([r[0]]) });
  }
}
// lines and symbols in Alaska and Hawaii get thinner strokes to match their smaller drawing
const BASE_W = { frontglow: 16, frontcase: 8.5, frontline: 4.4, frontalt: 4.4, frontsym: 1.6, extglow: 18, extline: 4.6, extsym: 1.5, extflake: 5, extray: 2.2, extsun: 1.4,
                 exthalo1: 15, exthalo2: 23, loccase: 7, locline: 4 };
function scaleStrokes(layer){
  layer.selectAll('path, circle').each(function(x){
    const cls = (this.getAttribute('class') || '').split(' ');
    const base = BASE_W[cls[0]] ?? null; if (base === null) return;
    const k = x && x.k && x.k < 1 ? x.k : 1;
    let w = base * k; if (cls[0] === 'extflake' && cls[1] === 'in') w = 2.4 * k;
    this.style.strokeWidth = k < 1 ? w : '';
    if (cls[0] === 'frontalt') this.style.strokeDasharray = k < 1 ? `${11*k} ${11*k}` : '';
    if (cls[0] === 'loccase' || cls[0] === 'locline') this.style.strokeDasharray = k < 1 ? `${14*k} ${8*k}` : '';
  });
}
function paintExtremes(TB){
  const out = { lines: [], syms: [], flakes: [], suns: [] }, fills = [];
  for (const [kind, on, thr] of [['arctic', showArctic, 0], ['heat', showHeat, 100]]){
    if (!on) continue;
    const v = TB.val(kind), d = TB.rate(kind);
    const rings = ringPaths(frw.gen.contour(v, thr));
    if (!rings.length) continue;
    const big = rings.reduce((m, r)=>Math.max(m, ringArea(r) / ([MIN_AREA, 160, 30][sample(frw.cellReg, r[0][0], r[0][1])] * 6)), 0);
    fills.push({ kind, d: toD(rings), op: Math.min(1, big) });
    extremeFront(rings, v, d, kind, out);
  }
  // ---- rare cold, judged against each place's own history (heat fronts are off the map) ----
  // rarity index 1 = coldest 1% of that place's days, 2 = coldest 0.25%
  const loc = { lines: [], syms: [], flakes: [], suns: [] }, locFills = [], cores = [], halos = [];
  frw.rareOn = { arctic: false };
  if (showArctic){
    const kind = 'arctic', key = 'rareC';
    const v = TB.val(key), d = TB.rate(key);
    const minA = r => [MIN_AREA, 160, 30][sample(frw.cellReg, r[0][0], r[0][1])];
    const r1 = ringPaths(frw.gen.contour(v, 1)).filter(r=>ringArea(r) >= 3*minA(r));     // no faded scraps
    const r2 = ringPaths(frw.gen.contour(v, 2));
    if (r1.length){
      frw.rareOn.arctic = true;
      locFills.push({ kind, d: toD(r1) });
      extremeFront(r1, v, d, kind, loc);                     // Local Arctic front: dashed, outside the official front
    }
    if (r2.length) cores.push({ kind, d: toD(r2) });         // the rarest 0.25%: a soft brighter patch, no line
    // official front: an even glow along the stretches where that cold is rare too (wider where it is the rarest)
    for (const l of out.lines){
      if (l.type !== kind || !l.pts) continue;
      for (const [t, lim] of [[1, 1], [2, 2]]){
        let run = [];
        const flush = ()=>{ if (run.length >= 3) halos.push({ t, d: lineD(run), k: l.k, op: l.op }); run = []; };
        for (const p of l.pts){ if (sample(v, p[0], p[1]) >= lim) run.push(p); else flush(); }
        flush();
      }
    }
  }
  d3.select('#offMaskPath').attr('d', fills.map(x=>x.d).join(''));
  d3.select('#offMaskEdge').attr('d', out.lines.map(l=>l.d).join(''));
  locFill.selectAll('path.locfill').data(locFills).join('path').attr('class', x=>'locfill '+x.kind).attr('fill-rule','evenodd').attr('d', x=>x.d);
  locHalo.selectAll('path').data([]).join('path');
  locLines.selectAll('path.loccase').data(loc.lines).join('path').attr('class','loccase').attr('d', x=>x.d);
  locLines.selectAll('path.locline').data(loc.lines).join('path').attr('class', x=>'locline '+x.type).attr('d', x=>x.d);
  locSyms.selectAll('path.extsym').data(loc.syms).join('path').attr('class', x=>'extsym '+x.type).attr('d', x=>x.d);
  locSyms.selectAll('path.extflake').data(loc.flakes.flatMap(f=>[{d: f.d, k: f.k, op: f.op, c:'extflake'}, {d: f.d, k: f.k, op: f.op, c:'extflake in'}])).join('path').attr('class', x=>x.c).attr('d', x=>x.d);
  extCore.selectAll('path').data(cores).join('path').attr('class', x=>'rarecore '+x.kind).attr('fill-rule','evenodd').attr('d', x=>x.d);
  haloLayer.selectAll('path').data(halos.sort((a,b)=>a.t-b.t)).join('path').attr('class', x=>`exthalo${x.t} arctic`).attr('d', x=>x.d);
  for (const L of [locLines, locSyms, haloLayer]){
    scaleStrokes(L);
    L.selectAll('path, circle').attr('opacity', x=>x && x.op !== undefined && x.op < 1 ? x.op : null);
  }
  frw.extOn = { arctic: fills.some(x=>x.kind==='arctic'), heat: fills.some(x=>x.kind==='heat') };
  frw.haloOn = halos.length;
  extFill.selectAll('path').data(fills).join('path').attr('class', x=>'extfill '+x.kind).attr('fill-rule','evenodd').attr('d', x=>x.d);
  extGlow.selectAll('path.extglow').data(out.lines).join('path').attr('class', x=>'extglow '+x.type).attr('d', x=>x.d);
  extLines.selectAll('path.frontcase').data(out.lines).join('path').attr('class','frontcase').attr('d', x=>x.d);
  extLines.selectAll('path.extsym').data(out.syms).join('path').attr('class', x=>'extsym '+x.type).attr('d', x=>x.d);
  extLines.selectAll('path.extline').data(out.lines).join('path').attr('class', x=>'extline '+x.type).attr('d', x=>x.d);
  extLines.selectAll('path.extflake').data(out.flakes.flatMap(f=>[{d: f.d, k: f.k, op: f.op, c:'extflake'}, {d: f.d, k: f.k, op: f.op, c:'extflake in'}])).join('path').attr('class', x=>x.c).attr('d', x=>x.d);
  extLines.selectAll('path.extray').data(out.suns).join('path').attr('class','extray').attr('d', x=>x.rays);
  extLines.selectAll('circle.extsun').data(out.suns).join('circle').attr('class','extsun').attr('cx', x=>x.c[0]).attr('cy', x=>x.c[1]).attr('r', x=>7*x.k);
  scaleStrokes(extLines); scaleStrokes(extGlow);
  extGlow.selectAll('path').attr('opacity', x=>x && x.op !== undefined && x.op < 1 ? x.op : null);
  extLines.selectAll('path, circle').attr('opacity', x=>x && x.op !== undefined && x.op < 1 ? x.op : null);
  extFill.selectAll('path').attr('opacity', x=>x.op < 1 ? x.op : null);
}
// ----- Jan 1 / Jul 1 turnover: the old spring or autumn color fades out over the new one
let ripple = null, lastFrontDate = null;
const reducedMotion = (()=>{ try { return matchMedia('(prefers-reduced-motion: reduce)').matches; } catch(e){ return false; } })();

function startRipple(oldLabel){
  if (reducedMotion || !frw) return;
  ripple = { t0: performance.now(), dur: 500, old: oldLabel, op: 1 };
  paintFronts();
  requestAnimationFrame(rippleStep);
}
function rippleStep(now){
  if (!ripple) return;
  const k = Math.min(1, (now - ripple.t0) / ripple.dur);
  ripple.op = 1 - k*k*(3-2*k);
  fillPaths.selectAll('path').filter(x=>x[0]==='ripple').attr('opacity', ripple.op);
  if (k < 1) requestAnimationFrame(rippleStep);
  else { ripple = null; paintFronts(); }
}
function checkTurnover(iso){
  if (lastFrontDate && lastFrontDate.slice(0,4) !== iso.slice(0,4)) startRipple(iso > lastFrontDate ? 'autumn' : 'spring');
  lastFrontDate = iso;
}

// Smooth motion through time: each field follows a curve through the days before and after (Catmull-Rom), so the
// fronts move at a steady pace instead of slowing to a stop at every day. Values stay between the two days' values.
function timeBlend(off, t){
  const cl = o => Math.max(0, Math.min(frw.maxOff, o));
  const F = [frFields(cl(off-1)), frFields(cl(off)), frFields(cl(off+1)), frFields(cl(off+2))];
  const cells = frw.nx*frw.ny, memo = {}, dmemo = {};
  const t2 = t*t, t3 = t2*t;
  return {
    F,
    val(name){
      if (memo[name]) return memo[name];
      const [a, b, c, d] = F.map(x=>x[name]), out = new Float64Array(cells);
      for (let q=0;q<cells;q++){
        const p0 = a[q], p1 = b[q], p2 = c[q], p3 = d[q];
        let v = 0.5*(2*p1 + (p2-p0)*t + (2*p0 - 5*p1 + 4*p2 - p3)*t2 + (3*p1 - p0 - 3*p2 + p3)*t3);
        const lo = p1 < p2 ? p1 : p2, hi = p1 < p2 ? p2 : p1;
        out[q] = v < lo ? lo : v > hi ? hi : v;
      }
      return memo[name] = out;
    },
    rate(name){       // change per day, blended smoothly so front types do not flip right at midnight
      if (dmemo[name]) return dmemo[name];
      const [a, b, c, d] = F.map(x=>x[name]), out = new Float64Array(cells);
      for (let q=0;q<cells;q++) out[q] = (1-t)*(c[q]-a[q])/2 + t*(d[q]-b[q])/2;
      return dmemo[name] = out;
    }
  };
}
let prefetchTimer = null;
function prefetchAhead(off){      // work out the next few days in spare moments, so a new day never causes a hitch
  if (prefetchTimer) return;
  const idle = fn => (window.requestIdleCallback ? requestIdleCallback(fn, { timeout: 120 }) : setTimeout(fn, 30));
  const todo = [off+3, off+4, off+2].filter(o=>o >= 0 && o <= frw.maxOff && !frCache.has(o));
  const iso = o => new Date(frw.d0 + o*864e5).toISOString().slice(0,10);
  const wtodo = (showWatches && typeof watchesFor === 'function')
    ? [off+1, off+2].map(iso).filter(d=>WLOG[d] && !boxCache.has(d)) : [];
  if (!todo.length && !wtodo.length) return;
  prefetchTimer = idle(()=>{
    prefetchTimer = null;
    if (todo.length) frFields(todo[0]); else watchesFor(wtodo[0]);
    prefetchAhead(off);
  });
}

function paintFronts(){
  const off = Math.round((Date.parse(curISO()+'T00:00:00Z') - frw.d0)/864e5);
  const TB = timeBlend(off, frac), f = frac;
  const [, A, B] = TB.F;
  const cells = frw.nx*frw.ny;
  const tier = TB.val('tier'), mild = TB.val('mild'), dT = TB.rate('tier');
  const neg = new Float64Array(cells); for (let q=0;q<cells;q++) neg[q] = -tier[q];
  prefetchAhead(off);
  const springR = ringPaths(frw.gen.contour(mild, 0.0001));
  const winterR = ringPaths(frw.gen.contour(neg, -0.5));
  const summerR = ringPaths(frw.gen.contour(tier, 1.5));
  const allR = [[[0,0],[1000,0],[1000,640],[0,640]]];

  const out = { lines: [], syms: [] };
  frontPieces(winterR, tier, dT, out);

  const px = frw.img.data, tempB = TB.val('temp');
  for (let q=0;q<cells;q++){
    const tq = tempB[q];
    const g8 = Math.round(255 * Math.max(0.06, Math.min(0.94, 0.5 + (tq - 55) / 80)));
    px[4*q] = px[4*q+1] = px[4*q+2] = g8; px[4*q+3] = 255;
  }
  frw.ctx.putImageData(frw.img, 0, 0);
  tempImg.attr('width', frw.nx*frw.SK).attr('height', frw.ny*frw.SK).attr('href', frw.cv.toDataURL());
  lineLayer.selectAll('path.frontglow').data(out.lines).join('path').attr('class', x=>'frontglow '+x.type).attr('d', x=>x.d);
  lineLayer.selectAll('path.frontcase').data(out.lines).join('path').attr('class','frontcase').attr('d', x=>x.d);
  lineLayer.selectAll('path.frontsym').data(out.syms).join('path').attr('class', x=>'frontsym '+x.type).attr('d', x=>x.d);
  lineLayer.selectAll('path.frontline').data(out.lines).join('path').attr('class', x=>'frontline '+x.type).attr('d', x=>x.d);
  lineLayer.selectAll('path.frontalt').data(out.lines.filter(x=>x.type==='still')).join('path').attr('class','frontalt').attr('d', x=>x.d);
  scaleStrokes(lineLayer);
  paintExtremes(TB);
  shapes.attr('fill', 'var(--autumn)');
  const dd = new Date(curISO()+'T00:00:00Z');
  stampDate.text(`${MON[dd.getUTCMonth()].toUpperCase()} ${dd.getUTCDate()}`);
  stampYear.text(`${dd.toLocaleDateString('en-US',{weekday:'long',timeZone:'UTC'}).toUpperCase()} · ${dd.getUTCFullYear()}`);
  outlookLook(curISO(), frac);
}

function rdp(pts, eps){                 // Douglas-Peucker: straight-edged outline
  if (pts.length < 4) return pts;
  const [a, b] = [pts[0], pts[pts.length-1]];
  let idx = -1, best = 0;
  for (let i=1;i<pts.length-1;i++){
    const p = pts[i], dx = b[0]-a[0], dy = b[1]-a[1], L = Math.hypot(dx,dy) || 1;
    const dist = Math.abs(dy*p[0] - dx*p[1] + b[0]*a[1] - b[1]*a[0]) / L;
    if (dist > best){ best = dist; idx = i; }
  }
  if (best <= eps) return [a, b];
  const l = rdp(pts.slice(0, idx+1), eps), r = rdp(pts.slice(idx), eps);
  return l.slice(0, -1).concat(r);
}
function simplifyRing(r){
  // split the closed ring at its two farthest-apart points so the simplification has fixed ends
  let far = 0, fi = 0;
  for (let i=1;i<r.length;i++){ const d = Math.hypot(r[i][0]-r[0][0], r[i][1]-r[0][1]); if (d > far){ far = d; fi = i; } }
  const a = rdp(r.slice(0, fi+1), 9), b = rdp(r.slice(fi).concat([r[0]]), 9);
  return a.slice(0, -1).concat(b.slice(0, -1));
}
function inPoly(x, y, poly){
  let inside = false;
  for (let i=0, j=poly.length-1; i<poly.length; j=i++){
    const xi=poly[i][0], yi=poly[i][1], xj=poly[j][0], yj=poly[j][1];
    if (((yi>y)!==(yj>y)) && (x < (xj-xi)*(y-yi)/(yj-yi)+xi)) inside = !inside;
  }
  return inside;
}
// ----- simple watch boxes: grouped places -> convex outline cut down to at most 6 sides, with a margin -----
const BOX = { watch: { link: 68, pad: 22, minPts: 2 }, warning: { link: 50, pad: 13, minPts: 1 }, maxSides: 6, maxPerSeason: 3 };
function convexHull(pts){
  const p = [...new Map(pts.map(q=>[q[0]+','+q[1], q])).values()].sort((a,b)=>a[0]-b[0] || a[1]-b[1]);
  if (p.length < 3) return p;
  const cr = (o,a,b)=>(a[0]-o[0])*(b[1]-o[1])-(a[1]-o[1])*(b[0]-o[0]);
  const lo = [], up = [];
  for (const q of p){ while (lo.length>=2 && cr(lo[lo.length-2], lo[lo.length-1], q) <= 0) lo.pop(); lo.push(q); }
  for (let i=p.length-1;i>=0;i--){ const q=p[i]; while (up.length>=2 && cr(up[up.length-2], up[up.length-1], q) <= 0) up.pop(); up.push(q); }
  return lo.slice(0,-1).concat(up.slice(0,-1));       // counter-clockwise in screen space? (y down -> clockwise visually)
}
function lineX(a1, a2, b1, b2){                        // intersection of line a1-a2 with line b1-b2
  const d = (a2[0]-a1[0])*(b2[1]-b1[1]) - (a2[1]-a1[1])*(b2[0]-b1[0]);
  if (Math.abs(d) < 1e-9) return null;
  const t = ((b1[0]-a1[0])*(b2[1]-b1[1]) - (b1[1]-a1[1])*(b2[0]-b1[0])) / d;
  return [a1[0] + t*(a2[0]-a1[0]), a1[1] + t*(a2[1]-a1[1])];
}
const triArea = (a,b,c)=>Math.abs((b[0]-a[0])*(c[1]-a[1]) - (b[1]-a[1])*(c[0]-a[0]))/2;
function reduceSides(poly, maxN){
  // drop one edge at a time by extending its two neighbours until they meet, always picking the edge that adds the
  // least area; the shape still contains everything it did before
  let p = poly.slice();
  while (p.length > maxN){
    let best = null;
    for (let i=0;i<p.length;i++){
      const a = p[(i-1+p.length)%p.length], b = p[i], c = p[(i+1)%p.length], d = p[(i+2)%p.length];
      const X = lineX(a, b, d, c); if (!X) continue;
      // X must lie beyond b (going a->b) and beyond c (going d->c)
      if ((X[0]-b[0])*(b[0]-a[0]) + (X[1]-b[1])*(b[1]-a[1]) < 0) continue;
      if ((X[0]-c[0])*(c[0]-d[0]) + (X[1]-c[1])*(c[1]-d[1]) < 0) continue;
      const add = triArea(b, X, c);
      if (!best || add < best.add) best = { i, X, add };
    }
    if (!best) break;
    const j = (best.i+1)%p.length;
    p[best.i] = best.X; p.splice(j, 1);
  }
  return p;
}
function offsetPoly(poly, pad){
  // move every edge outward by pad and rebuild the corners
  const n = poly.length; if (n < 3) return poly;
  let area = 0; for (let i=0;i<n;i++){ const a=poly[i], b=poly[(i+1)%n]; area += a[0]*b[1]-b[0]*a[1]; }
  const s = area > 0 ? 1 : -1;
  const lines = [];
  for (let i=0;i<n;i++){
    const a = poly[i], b = poly[(i+1)%n]; let nx = b[1]-a[1], ny = -(b[0]-a[0]); const L = Math.hypot(nx,ny)||1;
    nx = nx/L*pad*s; ny = ny/L*pad*s;
    lines.push([[a[0]+nx, a[1]+ny], [b[0]+nx, b[1]+ny]]);
  }
  const out = [];
  for (let i=0;i<n;i++){
    const l1 = lines[(i-1+n)%n], l2 = lines[i];
    const X = lineX(l1[0], l1[1], l2[0], l2[1]);
    out.push(X && Math.hypot(X[0]-poly[i][0], X[1]-poly[i][1]) < pad*4 ? X : l2[0]);
  }
  return out;
}
function smallBox(pts, pad){
  // one or two places: a tilted rectangle along the pair (or a slightly tilted square)
  const [a, b] = [pts[0], pts[pts.length-1]];
  let ang = pts.length > 1 ? Math.atan2(b[1]-a[1], b[0]-a[0]) : -0.26;
  const ca = Math.cos(ang), sa = Math.sin(ang), mx = (a[0]+b[0])/2, my = (a[1]+b[1])/2;
  const half = Math.hypot(b[0]-a[0], b[1]-a[1])/2 + pad + 6, hw = pad + 8;
  return [[-half,-hw],[half,-hw],[half,hw],[-half,hw]].map(([u,v])=>[mx + u*ca - v*sa, my + u*sa + v*ca]);
}
function groupBy(pts, link){
  const parent = pts.map((_, i)=>i), find = i=>{ while (parent[i]!==i){ parent[i]=parent[parent[i]]; i=parent[i]; } return i; };
  for (let i=0;i<pts.length;i++) for (let j=i+1;j<pts.length;j++)
    if ((pts[i].x-pts[j].x)**2 + (pts[i].y-pts[j].y)**2 <= link*link) parent[find(i)] = find(j);
  const g = new Map(); pts.forEach((p, i)=>{ const r = find(i); if (!g.has(r)) g.set(r, []); g.get(r).push(p); });
  return [...g.values()];
}
function boxFor(gp, pad){
  const xy = gp.map(p=>[p.x, p.y]);
  const hull = convexHull(xy);
  let box = hull.length >= 3 ? offsetPoly(reduceSides(hull, BOX.maxSides), pad) : smallBox(xy, pad);
  if (hull.length >= 3 && ringArea(box) < 1800) box = smallBox([xy.reduce((a,p)=>p[0]<a[0]?p:a), xy.reduce((a,p)=>p[0]>a[0]?p:a)], pad);
  return box;
}
// Split a group (tighter grouping each time) until at least 60% of the checked places inside its box are changing.
function splitGroup(gp, link, pad, eligiblePts){
  const box = boxFor(gp, pad);
  const inside = eligiblePts.filter(q=>inPoly(q[0], q[1], box)).length;
  if (gp.length / Math.max(inside, 1) >= 0.6 || link < 28 || gp.length < 3) return [{ gp, box }];
  return groupBy(gp, link*0.75).flatMap(sub=>splitGroup(sub, link*0.75, pad, eligiblePts));
}
function simpleBoxes(byPoint, P, eligiblePts){
  const boxes = [];
  for (const level of ['watch', 'warning']){
    for (const season of ['winter', 'spring', 'summer', 'autumn']){
      const pts = [];
      for (const [i, e] of byPoint) if (e[2] === season && (level === 'watch' || e[1] === 'warning')) pts.push({ x: P[i][0], y: P[i][1], e });
      if (!pts.length) continue;
      const cfg = BOX[level];
      const groups = groupBy(pts, cfg.link).flatMap(g=>splitGroup(g, cfg.link, cfg.pad, eligiblePts))
        .filter(({ gp })=>gp.length >= cfg.minPts || gp.some(p=>isCapitalKey(p.e[0])));
      groups.sort((a,b)=>b.gp.length - a.gp.length);
      for (const { gp, box } of groups.slice(0, BOX.maxPerSeason)){
        const m = gp.map(p=>[p.e[0], p.e[3], p.e[4]]);
        m.sort((a,b)=> (isCapitalKey(b[0]) - isCapitalKey(a[0])) || a[1].localeCompare(b[1]) || a[0].localeCompare(b[0]));
        boxes.push({ s: season, lv: level, box: box.map(q=>[Math.round(q[0]), Math.round(q[1])]), m });
      }
    }
  }
  return boxes;
}

// ----- county-based watches: each county joins if the blend of nearby forecast places says the change is coming -----
const CTOPO = (()=>{ try { return JSON.parse(document.getElementById('counties').textContent); } catch(e){ return null; } })();
const CYCLE_NEXT = { spring: 'summer', summer: 'autumn', autumn: 'winter', winter: 'spring' };
let CO = null;
function countyInit(){
  if (CO || !CTOPO || !frw) return CO;
  const geoms = CTOPO.objects.counties.geometries;
  const feats = topojson.feature(CTOPO, CTOPO.objects.counties).features;
  const nb = topojson.neighbors(geoms);
  const cent = feats.map(f=>path.centroid(f));
  const arcsOf = g => (g.type === 'Polygon' ? g.arcs : g.arcs.flat()).flat();
  const use = new Map();
  for (const gm of geoms) for (const a of arcsOf(gm)){ const k = a < 0 ? ~a : a; use.set(k, (use.get(k)||0) + 1); }
  const edge = geoms.map(gm=>arcsOf(gm).some(a=>use.get(a < 0 ? ~a : a) === 1));   // touches the coast or border
  const { P, n } = frw, C = feats.length, W = new Float32Array(C*n);
  const creg = geoms.map(g=>regionOfFips(String(g.id)));
  for (let c=0;c<C;c++){ const [x,y] = cent[c]; for (let p=0;p<n;p++){
    if (frw.regPt[p] !== creg[c]) continue;
    const dx=P[p][0]-x, dy=P[p][1]-y, d2 = dx*dx+dy*dy;
    if (creg[c] === 1 && d2 > 50*50) continue;             // Alaska: a place only speaks for boroughs within ~400 miles
    W[c*n+p] = Math.pow(d2+400, -1.25); } }
  const ptCounty = P.map((p, i)=>{ let b=-1, bd=Infinity; for (let c=0;c<C;c++){ if (creg[c] !== frw.regPt[i]) continue; const d=(cent[c][0]-p[0])**2+(cent[c][1]-p[1])**2; if (d<bd){ bd=d; b=c; } } return b; });
  CO = { geoms, feats, nb, cent, edge, W, C, ptCounty, creg };
  return CO;
}
// size rules scale down for Alaska's and Hawaii's much bigger counties
const minCounties = (comp, base) => Math.max(1, Math.round(base / REGION_SCALE[CO.creg[comp[0]]]));
const topPerRegion = (comps, k) => { const cnt = [0, 0, 0]; return comps.filter(cp=>cnt[CO.creg[cp[0]]]++ < k); };
function components(set, nb){
  const seen = new Set(), comps = [];
  for (const s of set){
    if (seen.has(s)) continue;
    const comp = [s], st = [s]; seen.add(s);
    while (st.length){ const a = st.pop(); for (const b of nb[a]) if (set.has(b) && !seen.has(b)){ seen.add(b); comp.push(b); st.push(b); } }
    comps.push(comp);
  }
  return comps;
}
function bfsDist(sources, passable, nb, cap){
  const dist = new Map(); const q = [];
  for (const s of sources){ dist.set(s, 0); q.push(s); }
  for (let h=0; h<q.length; h++){
    const a = q[h], da = dist.get(a); if (da >= cap) continue;
    for (const b of nb[a]) if (!dist.has(b) && passable(b)){ dist.set(b, da+1); q.push(b); }
  }
  return dist;
}
function countyAreas(byPoint, eligible, shownAtCounty, shownAtPoint, exclude){
  const co = countyInit(); if (!co) return [];
  const { P, n } = frw, { C, W, nb, edge, geoms, ptCounty } = co;
  const out = [], watchSet = {};
  for (const level of ['watch', 'warning', 'exception']){
    const passes = level === 'exception' ? [['winter','latefreeze'], ['summer','summerreturns'], ['winter','earlyfreeze'], ['summer','earlysummer']]
                                         : ['winter', 'spring', 'summer', 'autumn'].map(s2=>[s2, null]);
    for (const [season, exKind] of passes){
      const ind = new Uint8Array(n), chg = new Uint8Array(n); let any = false;
      for (const [i, e] of byPoint){
        if (e[2] !== season || (e[5] || null) !== exKind) continue;
        chg[i] = 1;
        if (level !== 'warning' || e[1] === 'warning'){ ind[i] = 1; any = true; }
      }
      if (!any) continue;
      // 1) county decision from the blend of nearby places. Watches: among all checked places.
      //    Warnings: among the places forecast to change, and only inside that season's watch.
      const ex = (exclude && exclude[season]) || null;
      const inS = new Set();
      const purge = ()=>{ if (ex) for (const c of ex) inS.delete(c); };
      for (let c=0;c<C;c++){
        if (shownAtCounty[c] === season || (ex && ex.has(c))) continue;
        if (level === 'warning' && !(watchSet[season] && watchSet[season].has(c))) continue;
        let num=0, den=0; const row=c*n;
        for (let p=0;p<n;p++){
          if (!eligible[p] || (level === 'warning' && !chg[p])) continue;
          const w = W[row+p]; den += w; if (ind[p]) num += w;
        }
        if (den && num/den >= 0.4) inS.add(c);
      }
      if (!inS.size) continue;
      const fillHoles = ()=>{
        for (let pass=0; pass<2; pass++) for (let c=0;c<C;c++)
          if (!inS.has(c) && nb[c].length && nb[c].every(b=>inS.has(b)) && shownAtCounty[c] !== season) inS.add(c);
        const outside = new Set(); for (let c=0;c<C;c++) if (!inS.has(c)) outside.add(c);
        for (const comp of components(outside, nb)) if (comp.length <= 60 && !comp.some(c=>edge[c])) for (const c of comp) if (shownAtCounty[c] !== season) inS.add(c);
        purge();
      };
      fillHoles();
      // bridge: areas within about 4 counties (~100 miles) of each other join, filling the counties between them
      const passable = c=>shownAtCounty[c] !== season;
      for (let round=0; round<3; round++){
        const comps0 = components(inS, nb); if (comps0.length < 2) break;
        let joined = false;
        const dists = comps0.map(cp=>bfsDist(cp, passable, nb, 4));
        for (let a=0; a<comps0.length; a++) for (let b=a+1; b<comps0.length; b++){
          let dmin = Infinity; for (const c of comps0[b]) if (dists[a].has(c)) dmin = Math.min(dmin, dists[a].get(c));
          if (!(dmin <= 4)) continue;
          for (const [c, da] of dists[a]) if (dists[b].has(c) && da + dists[b].get(c) <= dmin + 1) inS.add(c);
          joined = true;
        }
        if (!joined) break;
        fillHoles();
      }
      // 2) extend each watch area to the front it comes from (the counties already showing that season)
      if (level === 'watch'){   // (not for exceptions: those are setbacks, not a front arriving)
        const srcs = []; for (let c=0;c<C;c++) if (shownAtCounty[c] === season) srcs.push(c);
        if (srcs.length){
          const distS = bfsDist(srcs, c=>shownAtCounty[c] !== season, nb, 12);
          for (const comp of components(inS, nb)){
            let dmin = Infinity; for (const c of comp) if (distS.has(c)) dmin = Math.min(dmin, distS.get(c));
            if (!(dmin >= 1 && dmin <= 8)) continue;
            const dA = bfsDist(comp, c=>shownAtCounty[c] !== season, nb, 12);
            for (const [c, da] of dA) if (distS.has(c) && distS.get(c) + da <= dmin + 1) inS.add(c);
          }
          fillHoles();
        }
      }
      purge();
      if (level === 'watch') watchSet[season] = new Set(inS);
      // 3) drop tiny areas, keep the 3 biggest
      const comps = components(inS, nb).filter(cp=>cp.length >= minCounties(cp, level === 'watch' ? 5 : 3)).sort((a,b)=>b.length - a.length);
      for (const comp of topPerRegion(comps, 3)){
        const cs = new Set(comp);
        const m = [];
        for (const [i, e] of byPoint) if (ind[i] && cs.has(ptCounty[i])) m.push([e[0], e[3], e[4], shownAtPoint(i)]);
        if (!m.length) continue;
        m.sort((a,b)=> (isCapitalKey(b[0]) - isCapitalKey(a[0])) || a[1].localeCompare(b[1]) || a[0].localeCompare(b[0]));
        const geom = topojson.merge(CTOPO, comp.map(c=>geoms[c]));
        const coords = geom.coordinates.flatMap(poly=>poly[0]);
        const top = coords.reduce((a,p)=>p[1]<a[1]?p:a, coords[0]);
        const hullPts = reduceSides(convexHull(coords), 8);
        const counties = comp.map(c=>[geoms[c].id, geoms[c].properties.n]);
        out.push({ s: season, lv: level === 'exception' ? 'warning' : level, ex: exKind, d: path(geom), top, box: hullPts, m, counties, ci: comp });
      }
    }
  }
  return out;
}

function rareAreas(rare, eligible){
  const co = countyInit(); if (!co || !rare || !rare.length) return [];
  const { P, n } = frw, { C, W, nb, edge, geoms, ptCounty } = co;
  const byType = {};
  for (const r of rare){ const i = keyIdx.get(r[0]); if (i !== undefined && eligible[i]) (byType[r[1]] = byType[r[1]] || new Map()).set(i, r); }
  const out = [];
  for (const type of Object.keys(RARE)){
    const pts = byType[type]; if (!pts) continue;
    const inS = new Set();
    for (let c=0;c<C;c++){
      let num=0, den=0; const row=c*n;
      for (let p=0;p<n;p++){ if (!eligible[p]) continue; const w = W[row+p]; den += w; if (pts.has(p)) num += w; }
      if (den && num/den >= 0.4) inS.add(c);
    }
    const fill = ()=>{
      for (let c=0;c<C;c++) if (!inS.has(c) && nb[c].length && nb[c].every(b=>inS.has(b))) inS.add(c);
      const outside = new Set(); for (let c=0;c<C;c++) if (!inS.has(c)) outside.add(c);
      for (const comp of components(outside, nb)) if (comp.length <= 60 && !comp.some(c=>edge[c])) for (const c of comp) inS.add(c);
    };
    fill();
    for (let round=0; round<3; round++){
      const cs = components(inS, nb); if (cs.length < 2) break;
      const ds = cs.map(cp=>bfsDist(cp, ()=>true, nb, 4)); let joined = false;
      for (let a=0;a<cs.length;a++) for (let b=a+1;b<cs.length;b++){
        let dmin = Infinity; for (const c of cs[b]) if (ds[a].has(c)) dmin = Math.min(dmin, ds[a].get(c));
        if (!(dmin <= 4)) continue;
        for (const [c, da] of ds[a]) if (ds[b].has(c) && da + ds[b].get(c) <= dmin + 1) inS.add(c);
        joined = true;
      }
      if (!joined) break; fill();
    }
    const comps = components(inS, nb).filter(cp=>cp.length >= minCounties(cp, 5)).sort((a,b)=>b.length - a.length);
    for (const comp of topPerRegion(comps, 3)){
      const cs = new Set(comp), m = [];
      for (const [i, r] of pts) if (cs.has(ptCounty[i])) m.push([r[0], r[2], r[3], r[4]]);
      if (!m.length) continue;
      m.sort((a,b)=> (isCapitalKey(b[0]) - isCapitalKey(a[0])) || a[1].localeCompare(b[1]) || a[0].localeCompare(b[0]));
      const geom = topojson.merge(CTOPO, comp.map(c=>geoms[c]));
      const coords = geom.coordinates.flatMap(poly=>poly[0]);
      out.push({ s: type, lv: 'advisory', d: path(geom), top: coords.reduce((a,p)=>p[1]<a[1]?p:a, coords[0]),
                 box: reduceSides(convexHull(coords), 8), m });
    }
  }
  return out;
}
// Rare cold / heat areas: counties join where the blend of nearby checked places says the forecast reaches that
// place's own tier-1 line (advisory) or tier-2 line (warning, drawn inside the advisory).
// Log entries: [key, type, first day, value, last day, tier, tier-1 line, tier-2 line]
function rareTempAreas(rareList, eligible){
  const co = countyInit(); if (!co) return [];
  const { n } = frw, { C, W, nb, edge, geoms, ptCounty } = co;
  const out = [];
  for (const type of Object.keys(RARE_TEMP)){
    const pts = new Map();
    for (const r of rareList){ if (r[1] !== type) continue; const i = keyIdx.get(r[0]); if (i !== undefined && eligible[i]) pts.set(i, r); }
    if (!pts.size) continue;
    let inT1 = null;
    for (const tier of [1, 2]){
      const flagged = new Map([...pts].filter(([, r])=>(r[5] || 1) >= tier));
      if (!flagged.size) continue;
      const inS = new Set();
      for (let c=0;c<C;c++){
        if (tier === 2 && !inT1.has(c)) continue;
        let num=0, den=0; const row=c*n;
        for (let p=0;p<n;p++){ if (!eligible[p]) continue; const w = W[row+p]; den += w; if (flagged.has(p)) num += w; }
        if (den && num/den >= 0.4) inS.add(c);
      }
      // every flagged place's own county counts, so a lone rare spot still shows
      for (const i of flagged.keys()) if (ptCounty[i] >= 0 && (tier === 1 || inT1.has(ptCounty[i]))) inS.add(ptCounty[i]);
      for (let c=0;c<C;c++) if (!inS.has(c) && nb[c].length && nb[c].every(b=>inS.has(b)) && (tier === 1 || inT1.has(c))) inS.add(c);
      const outside = new Set(); for (let c=0;c<C;c++) if (!inS.has(c)) outside.add(c);
      for (const comp of components(outside, nb)) if (comp.length <= 40 && !comp.some(c=>edge[c])) for (const c of comp) if (tier === 1 || inT1.has(c)) inS.add(c);
      if (tier === 1) inT1 = inS;
      const comps = components(inS, nb).sort((a,b)=>b.length - a.length);
      for (const comp of topPerRegion(comps, 10)){
        const cs = new Set(comp), m = [];
        for (const [i, r] of flagged) if (cs.has(ptCounty[i])) m.push([r[0], r[2], r[3], r[4], r[5] || 1, r[6], r[7]]);
        if (!m.length) continue;
        const rank = mm => RARE_TEMP[type].cold ? (mm[5] - mm[2]) : (mm[2] - mm[5]);     // degrees past its own line
        m.sort((a,b)=> (isCapitalKey(b[0]) - isCapitalKey(a[0])) || rank(b) - rank(a) || a[0].localeCompare(b[0]));
        const geom = topojson.merge(CTOPO, comp.map(c=>geoms[c]));
        const coords = geom.coordinates.flatMap(poly=>poly[0]);
        out.push({ s: type, lv: 'rare', tier, d: path(geom), rings: geom.coordinates, top: coords.reduce((a,p)=>p[1]<a[1]?p:a, coords[0]),
                   box: reduceSides(convexHull(coords), 8), m, ci: comp,
                   counties: comp.map(c=>[geoms[c].id, geoms[c].properties.n]) });
      }
    }
  }
  return out;
}
const boxCache = new Map();
// Boxes for one day: for each season, blend "this place is forecast to change to that season" across every point
// that was checked that morning (places already in that season, or not changing, count as no), outline where most
// nearby places agree, and straighten the outline. Warnings use only the confident places, so they sit inside watches.
function watchesFor(d){
  if (boxCache.has(d)) return boxCache.get(d);
  const raw = WLOG[d]; if (!raw || !frw) return null;
  const entries = Array.isArray(raw) ? raw : raw.e, all = Array.isArray(raw) ? false : !!raw.all;
  const off = Math.round((Date.parse(d+'T00:00:00Z') - frw.d0)/864e5);
  const { P, n, nx, ny, W, SK } = frw, cells = nx*ny;
  const eligible = new Uint8Array(n);
  for (let p=0;p<n;p++) eligible[p] = (P[p][2] <= off && off <= P[p][3] && (all || isCapitalKey(P[p][6] || ''))) ? 1 : 0;
  // the season the map is showing at each place that day; a box never counts places already shown in its season
  const F = frFields(off);
  const negT = F.tier.map(v=>-v);
  const R = { summer: ringPaths(frw.gen.contour(F.tier, 1.5)), winter: ringPaths(frw.gen.contour(negT, -0.5)),
              spring: ringPaths(frw.gen.contour(F.mild, 0.0001)) };
  const inRings = (rings, x, y)=>{ let c = 0; for (const r of rings) if (inPoly(x, y, r)) c++; return c % 2 === 1; };
  const shown = (x, y)=> inRings(R.summer, x, y) ? 'summer' : inRings(R.winter, x, y) ? 'winter' : inRings(R.spring, x, y) ? 'spring' : 'autumn';
  const byPoint = new Map();
  for (const e of entries){
    const i = keyIdx.get(e[0]);
    // only forward moves in the season cycle get alerts (no 'summer back to spring' and the like)
    const sh = i !== undefined ? shown(P[i][0], P[i][1]) : null;
    const exOk = ((e[5] === 'latefreeze' || e[5] === 'earlyfreeze') && sh === 'spring' && e[2] === 'winter') || ((e[5] === 'summerreturns' || e[5] === 'earlysummer') && sh === 'autumn' && e[2] === 'summer');
    if (i !== undefined && (CYCLE_NEXT[sh] === e[2] || exOk)) byPoint.set(i, e);
  }
  // ---- false seasons (a new season expected to flip back within a week) ----
  const RET = { falsespring:'winter', falsesummer:'spring', falseautumn:'summer', falsewinter:'autumn' };
  const PRETEND = { falsespring:'spring', falsesummer:'summer', falseautumn:'autumn', falsewinter:'winter' };
  const rareList = (raw && !Array.isArray(raw) && raw.r) ? raw.r : [];
  // a place expected to flip back never also counts toward the alert for flipping back
  const flips = new Map();
  for (const r of rareList){ if (!flips.has(r[0])) flips.set(r[0], []); flips.get(r[0]).push(r); }
  for (const [i, e] of [...byPoint]){
    const rs = flips.get(e[0]);
    if (rs && rs.some(r=> e[2] === RET[r[1]] && e[3] >= r[2] && e[3] <= r[4])) byPoint.delete(i);
  }
  let boxes, hitByType = {}, underway = {}, exclude = {}, shownAtCounty = null;
  if (countyInit()){
    shownAtCounty = CO.cent.map(c=>shown(c[0], c[1]));
    const { C, W, nb } = CO;
    for (const type of Object.keys(RARE)){
      const flagged = new Uint8Array(n); let any = false;
      for (const r of rareList){ if (r[1] !== type) continue; const i = keyIdx.get(r[0]); if (i !== undefined && eligible[i]){ flagged[i] = 1; any = true; } }
      if (!any) continue;
      const hit = new Uint8Array(C);
      for (let c=0;c<C;c++){
        let num=0, den=0; const row=c*n;
        for (let p=0;p<n;p++){ if (!eligible[p]) continue; const w = W[row+p]; den += w; if (flagged[p]) num += w; }
        if (den && num/den >= 0.4) hit[c] = 1;
      }
      hitByType[type] = hit;
      // already in the false season now: a stand-alone hatch, and kept out of the alert for flipping back
      const now = new Set(); for (let c=0;c<C;c++) if (hit[c] && shownAtCounty[c] === PRETEND[type]) now.add(c);
      const comps = components(now, nb).filter(cp=>cp.length >= minCounties(cp, 8));
      if (comps.length){
        underway[type] = comps;
        const ex = exclude[RET[type]] = exclude[RET[type]] || new Set();
        for (const cp of comps) for (const c of cp) ex.add(c);
      }
    }
    boxes = countyAreas(byPoint, eligible, shownAtCounty, i=>shown(P[i][0], P[i][1]), exclude);
  } else {
    const eligiblePts = [];
    for (let p=0;p<n;p++) if (eligible[p]) eligiblePts.push([P[p][0], P[p][1]]);
    boxes = simpleBoxes(byPoint, P, eligiblePts);
  }
  if (CO && shownAtCounty){
    const { nb, geoms, ptCounty } = CO;
    for (const type of Object.keys(hitByType)){
      const hit = hitByType[type];
      // coming soon: hatch the part of that season's watch or warning that will not last
      for (const bx of boxes){
        if (!bx.ci || bx.s !== PRETEND[type]) continue;
        const inside = new Set(bx.ci.filter(c=>hit[c]));
        const comps = components(inside, nb).filter(cp=>cp.length >= minCounties(cp, 8));
        if (!comps.length) continue;
        const geom = topojson.merge(CTOPO, comps.flat().map(c=>geoms[c]));
        (bx.hatches = bx.hatches || []).push({ type, d: path(geom), n: comps.reduce((s,cp)=>s+cp.length, 0) });
      }
      // already happening: a stand-alone hatch with its own bulletin
      for (const comp of (underway[type] || [])){
        const cs = new Set(comp), m = [];
        for (const r of rareList){ if (r[1] !== type) continue; const i = keyIdx.get(r[0]); if (i !== undefined && cs.has(ptCounty[i])) m.push([r[0], r[2], r[3], r[4]]); }
        if (!m.length) continue;
        m.sort((a,b)=> (isCapitalKey(b[0]) - isCapitalKey(a[0])) || a[1].localeCompare(b[1]) || a[0].localeCompare(b[0]));
        const geom = topojson.merge(CTOPO, comp.map(c=>geoms[c]));
        const coords = geom.coordinates.flatMap(poly=>poly[0]), dd = path(geom);
        boxes.push({ s: type, lv: 'advisory', d: dd, rings: geom.coordinates, top: coords.reduce((a,p)=>p[1]<a[1]?p:a, coords[0]),
                     box: reduceSides(convexHull(coords), 8), m, ci: comp, hatches: [{ type, d: dd, n: comp.length }], falseType: type });
      }
    }
    for (const bx of boxes) if (bx.hatches && bx.lv !== 'advisory'){ bx.hatches.sort((a,b)=>b.n - a.n); bx.falseType = bx.hatches[0].type; }
    boxes.push(...rareTempAreas(rareList, eligible).map(x=>({ ...x, rebuilt: !!raw.rb })));
    // ---- outlook areas: season changes the CFS outlook shows in weeks 2-4 (faint, never over a watch or warning) ----
    const olist = (raw && !Array.isArray(raw) && raw.o) ? raw.o : [];
    if (olist.length){
      const eligO = new Uint8Array(n);
      for (let p=0;p<n;p++) eligO[p] = (P[p][2] <= off && off <= P[p][3]) ? 1 : 0;
      const byO = new Map();
      for (const o of olist){
        const i = keyIdx.get(o[0]); if (i === undefined) continue;
        const sh = shown(P[i][0], P[i][1]);
        // the next season, or the one after when a watch is about to move the place along first
        if (CYCLE_NEXT[sh] === o[1] || CYCLE_NEXT[CYCLE_NEXT[sh]] === o[1]) byO.set(i, [o[0], 'watch', o[1], o[2], o[3]]);
      }
      const exO = {};
      for (const bx of boxes) if (bx.ci && (bx.lv === 'watch' || bx.lv === 'warning')){ const ex = exO[bx.s] = exO[bx.s] || new Set(); for (const c of bx.ci) ex.add(c); }
      const ob = countyAreas(byO, eligO, shownAtCounty, i=>shown(P[i][0], P[i][1]), exO)
        .filter(x=>x.lv === 'watch').map((x, k)=>({ ...x, lv: 'outlook', i: k }));
      boxes.unshift(...ob);
    }
  }
  boxCache.set(d, boxes);
  if (boxCache.size > 16) boxCache.delete(boxCache.keys().next().value);
  return boxes;
}

// ----- smooth alert transitions between day d and the next day -----
// An alert that continues sheds and gains counties one by one: a county the front passes drops the moment the
// front reaches it; other leaving counties peel off from the outside in, and new ones grow outward from the edge.
// An alert that does not continue fades out as a whole; a brand-new one fades in.
const trCache = new Map();
const FIRST_LOG = Object.keys(WLOG).sort()[0] || '9999';
function seasonAtFrac(FA, FB, x, y, e){
  const tv = sample(FA.tier, x, y)*(1-e) + sample(FB.tier, x, y)*e;
  if (tv <= 0.5) return 'winter';
  if (tv >= 1.5) return 'summer';
  return (sample(FA.mild, x, y)*(1-e) + sample(FB.mild, x, y)*e) > 0 ? 'spring' : 'autumn';
}
function pairPlan(a, b, d, FA, FB){
  const key = d + '|' + a.s + a.lv + (a.ex || '') + (a.tier || '') + '|' + a.ci.length + '|' + b.ci.length + '|' + a.ci[0] + '|' + b.ci[0];
  if (trCache.has(key)) return trCache.get(key);
  const { nb, cent } = CO;
  const A = new Set(a.ci), B = new Set(b.ci), core = a.ci.filter(c=>B.has(c));
  const drops = a.ci.filter(c=>!B.has(c)), adds = b.ci.filter(c=>!A.has(c));
  const dist = (sources, within)=>{ const m = new Map(), q = []; for (const s of sources){ m.set(s, 0); q.push(s); }
    for (let h=0; h<q.length; h++){ const c = q[h]; for (const n2 of nb[c]) if (within.has(n2) && !m.has(n2)){ m.set(n2, m.get(c)+1); q.push(n2); } } return m; };
  const dD = dist(core, A), dA = dist(core.length ? core : [b.ci[0]], B);
  const maxD = Math.max(1, ...drops.map(c=>dD.get(c) || 0)), maxA = Math.max(1, ...adds.map(c=>dA.get(c) || 0));
  const season = (a.lv === 'advisory' || a.lv === 'rare') ? null : a.s;
  const dropT = drops.map(c=>{
    let t = null;
    if (season) for (let k=1; k<=20; k++){ const e = k/20; if (seasonAtFrac(FA, FB, cent[c][0], cent[c][1], e) === season){ t = e; break; } }
    if (t === null) t = 0.85 - 0.7 * ((dD.get(c) ?? maxD) / maxD) * 0.999;   // outside first
    return [c, t];
  }).sort((p,q)=>p[1]-q[1]);
  const addT = adds.map(c=>[c, 0.15 + 0.7 * ((dA.get(c) ?? maxA) / maxA)]).sort((p,q)=>p[1]-q[1]);
  const plan = { core, dropT, addT, geo: new Map() };
  trCache.set(key, plan); if (trCache.size > 60) trCache.delete(trCache.keys().next().value);
  return plan;
}
function planGeometry(plan, e){
  let nd = 0; while (nd < plan.dropT.length && plan.dropT[nd][1] <= e) nd++;
  let na = 0; while (na < plan.addT.length && plan.addT[na][1] <= e) na++;
  const sig = nd + ':' + na;
  if (plan.geo.has(sig)) return plan.geo.get(sig);
  const cs = plan.core.concat(plan.dropT.slice(nd).map(p=>p[0]), plan.addT.slice(0, na).map(p=>p[0]));
  let out = null;
  if (cs.length){
    const geom = topojson.merge(CTOPO, cs.map(c=>CO.geoms[c]));
    const coords = geom.coordinates.flatMap(poly=>poly[0]);
    out = { d: path(geom), rings: geom.coordinates, top: coords.reduce((a2,p)=>p[1]<a2[1]?p:a2, coords[0]) };
  }
  plan.geo.set(sig, out);
  return out;
}
function transitionGroups(d){
  const A = watchesFor(d) || [];
  const base = A.map((x, i)=>({ ...x, i, key: d+'|'+i, op: 1 }));
  if (!(frac > 0) || !frw || !CO) return base;
  const nd = nextDay(d);
  if (nd > DATA.obsEnd || nd < FIRST_LOG) return base;
  const B = WLOG[nd] ? (watchesFor(nd) || []) : [];      // no log that day = no alerts: today's fade out
  const e = frac*frac*(3-2*frac);
  const off = Math.round((Date.parse(d+'T00:00:00Z') - frw.d0)/864e5);
  const FA = frFields(off), FB = frFields(Math.min(off+1, frw.maxOff));
  // match each of today's alerts to the same kind of alert tomorrow with the most shared counties
  const usedB = new Set(), out = [];
  A.forEach((a, i)=>{
    let best = -1, bestN = 0;
    if (a.ci) B.forEach((b, j)=>{
      if (usedB.has(j) || !b.ci || b.s !== a.s || b.lv !== a.lv || (b.ex || null) !== (a.ex || null) || (b.tier || 0) !== (a.tier || 0)) return;
      const S = new Set(a.ci); let n2 = 0; for (const c of b.ci) if (S.has(c)) n2++;
      if (n2 > bestN){ bestN = n2; best = j; }
    });
    if (best < 0){ out.push({ ...a, i, key: d+'|out|'+i, op: 1 - e }); return; }
    usedB.add(best);
    const b = B[best], plan = pairPlan(a, b, d, FA, FB), g = planGeometry(plan, e);
    if (!g) return;
    const src = e < 0.5 ? a : b;
    out.push({ ...src, i, key: d+'|pair|'+i, op: 1, d: g.d, rings: g.rings, top: g.top,
               hatches: src.hatches, falseType: src.falseType });
  });
  B.forEach((b, j)=>{ if (!usedB.has(j)) out.push({ ...b, i: A.length + j, key: d+'|in|'+j, op: e }); });
  // keep the drawing order: watches, then warnings, then hatches on top
  const pri = x => x.lv === 'outlook' ? -1 : x.lv === 'advisory' ? 3 : x.lv === 'rare' ? 1.5 + (x.tier || 1)/10 : x.lv === 'warning' ? 1 : 0;
  return out.sort((p,q)=>pri(p) - pri(q));
}

    gsel.select('rect').attr('x', r.x0).attr('y', r.y0).attr('width', w).attr('height', h).attr('fill', alertCol(x)).attr('display', clash ? 'none' : null);

  }
}
const pad2 = n => String(n).padStart(2,'0');
function bulletinTime(d, region){
  // issued the morning after the map date (the page's own run time for the latest day), in the area's own time zone
  let date, h, m;
  if (d === DATA.obsEnd){ const [dd, tt] = DATA.updated.split('T'); date = dd; [h, m] = tt.split(':').map(Number); }
  else { date = nextDay(d); h = 5; m = 56; }
  const y = +date.slice(0,4), sunday = (mo, nth)=>{ const f = new Date(Date.UTC(y, mo, 1)).getUTCDay(); return 1 + (7 - f) % 7 + 7*(nth-1); };
  const md = +date.slice(5,7)*100 + +date.slice(8,10);
  const dst = md >= 300 + sunday(2, 2) && md < 1100 + sunday(10, 1);       // 2nd Sunday of March to 1st Sunday of November
  let back = 0, zone = dst ? 'EDT' : 'EST';
  if (region === 1){ back = 4; zone = dst ? 'AKDT' : 'AKST'; }
  else if (region === 2){ back = dst ? 6 : 5; zone = 'HST'; }
  h -= back; if (h < 0){ h += 24; date = prevDay(date); }
  return { date, text: `${(h%12)||12}${pad2(m)} ${h>=12?'PM':'AM'} ${zone}` };
}
const prevDay = iso => { const x = new Date(iso+'T00:00:00Z'); x.setUTCDate(x.getUTCDate()-1); return x.toISOString().slice(0,10); };

function wrap(text, width=66){
  const out = []; let line = '';
  for (const word of text.split(' ')){
    if ((line + ' ' + word).trim().length > width){ out.push(line.trim()); line = word; } else line += ' ' + word;
  }
  if (line.trim()) out.push(line.trim());
  return out.join('\n');
}

// In the Fronts view the slider moves in fractions of a day and the map glides toward wherever it is dragged.
let glidePos = null, glideTarget = null, glideRaf = null;
function glideStep(){
  const diff = glideTarget - glidePos;
  glidePos = Math.abs(diff) < 0.02 ? glideTarget : glidePos + diff * 0.22;
  const day = Math.floor(glidePos), changed = day !== state.day;
  state.day = day; frac = glidePos - day;
  const done = glidePos === glideTarget;
  if (done) render(); else if (changed) renderLite(); else { paintFronts(); paintWatches(curISO()); }
  glideRaf = done ? null : requestAnimationFrame(glideStep);
}
slider.addEventListener('input',()=>{
  const v = +slider.value;
  if (view === 'fronts' && frw){
    if (state.playing) playBtn.click();
    if (glidePos === null || !glideRaf) glidePos = state.day + frac;
    glideTarget = v;
    if (!glideRaf) glideRaf = requestAnimationFrame(glideStep);
  } else { state.day = Math.round(v); frac = 0; render(); }
});

// How far into the future the map is (days past the last observed day, with the glide fraction)
function futureDays(d, f){ return Math.round((Date.parse(d+'T00:00:00Z') - Date.parse(DATA.obsEnd+'T00:00:00Z'))/864e5) + (f || 0); }
function outlookLook(d, f){
  const k = futureDays(d, f), fcLen = DATA.horizon || 10;
  // the CFS part fades in its veil and dims the fronts a little more the further out it goes
  const veilOp = k <= fcLen ? 0 : Math.min(0.32, 0.12 + (k - fcLen) / 60 * 0.2);
  veil.attr('opacity', veilOp);
  const dim = k <= fcLen ? 1 : Math.max(0.6, 1 - (k - fcLen) / 90 * 0.4);
  lineLayer.attr('opacity', dim < 1 ? dim : null); extLayer.attr('opacity', dim < 1 ? dim : null);
  stampTag.text(k <= 0 ? '' : k <= fcLen ? 'FORECAST' : 'CFS OUTLOOK');
}

  const clamp = v => Math.max(0, Math.min(100, v));
  const a = clamp(pos(nextDay(DATA.obsEnd))), b = clamp(pos(DATA.fcEnd || DATA.obsEnd)), c = clamp(pos(LAST));
  const [fcz, cfz] = band.children;
  fcz.style.left = a + '%'; fcz.style.width = Math.max(0, b - a) + '%';
  cfz.style.left = b + '%'; cfz.style.width = Math.max(0, c - b) + '%';
  band.hidden = !(mx > 0) || c <= 0 || a >= 100;
}
function setYear(y, keepDay){
  const prevMonthDay = curISO().slice(5);
  state.year = y;
  const mx = maxDay(y); slider.max = mx;
  let d = Math.round((new Date(`${y}-${prevMonthDay}T00:00:00Z`)-utc(y,0,1))/864e5);
  if (isNaN(d)) d = 0;
  state.day = keepDay===undefined ? Math.min(d,mx) : Math.min(keepDay,mx);
  slider.value = state.day;
  paintFutureBand(y, mx);
  for (const b of yearsEl.children) b.setAttribute('aria-pressed', b.textContent==y);
  const months = document.getElementById('months'); months.innerHTML='';
  for (let m=0;m<12;m++){
    const dd = (utc(y,m,1)-utc(y,0,1))/864e5; if (dd>mx) break;
    if (window.innerWidth<520 && m%3) continue;
    const s=document.createElement('span'); s.textContent=MON[m]; s.style.left = (dd/mx*100)+'%';
    if (dd/mx > .94) s.style.transform='translateX(-100%)';
    months.appendChild(s);
  }
  renderTable(); render();
}

function forecastStrip(fc){
  if (!fc || !fc.length) return '';
  return `<p class="eyebrow" style="margin-top:14px">Next ${fc.length} days · high / low °F</p><div class="fc">${fc.map(([d,h,l])=>{
    const w = new Date(d+'T00:00:00Z').toLocaleDateString('en-US',{weekday:'short',timeZone:'UTC'});
    return `<div>${w} ${+d.slice(8)}<b>${h}°</b><span class="lo">${l}°</span></div>`; }).join('')}</div>`;
}

      let last = 0;

        if (!state.playing) return;
        const dtMs = last ? Math.min(ts-last, 100) : 0; last = ts;
        frac += dtMs/170;
        let dayChanged = false;
        while (frac >= 1){ frac -= 1; state.day++; dayChanged = true; }
        if (state.day >= +slider.max){ state.day = +slider.max; frac = 0; slider.value = state.day; render(); playBtn.click(); return; }
        slider.value = state.day + frac;
        if (dayChanged) renderLite(); else { paintFronts(); paintWatches(curISO()); }
        timer = requestAnimationFrame(tick);
      };
      timer = requestAnimationFrame(tick);
    } else {
      timer = setInterval(()=>{ state.day++; if (state.day > +slider.max){ state.day = +slider.max; playBtn.click(); } slider.value=state.day; render(); }, 55);
    }
  } else { clearInterval(timer); cancelAnimationFrame(timer); frac = 0; render(); }
};
